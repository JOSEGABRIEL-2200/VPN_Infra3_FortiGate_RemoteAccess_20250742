! Running-config del R-USUARIOS - Infraestructura 3 - Jose Gabriel Feliz Maria 2025-0742
! Nota: configuracion en ejecucion del R-USUARIOS en formato IOS, armada a partir de
! los comandos aplicados (scripts/R-USUARIOS_config.txt) y de lo verificado en el
! laboratorio (show ip dhcp binding: 10.7.42.10 y 10.7.42.11).
! Las contrasenas aparecen como <cifrada> porque IOS las guarda cifradas.
!
version 15.7
service timestamps debug datetime msec
service timestamps log datetime msec
service password-encryption
!
hostname R-USUARIOS
!
enable secret 5 <cifrada>
!
no aaa new-model
!
ip dhcp excluded-address 10.7.42.1 10.7.42.9
!
ip dhcp pool USUARIOS-VLAN10
 network 10.7.42.0 255.255.255.128
 default-router 10.7.42.1
 dns-server 8.8.8.8 8.8.4.4
!
no ip domain lookup
ip domain name itla.lab
ip cef
no ipv6 cef
!
username admin privilege 15 secret 5 <cifrada>
!
interface Ethernet0/0
 description WAN-ISP (IP publica sitio Usuarios)
 ip address 200.7.42.2 255.255.255.252
 ip nat outside
 ip virtual-reassembly in
 duplex auto
!
interface Ethernet0/1
 description LAN-USUARIOS (VLAN 10 via SW-A)
 ip address 10.7.42.1 255.255.255.128
 ip nat inside
 ip virtual-reassembly in
 duplex auto
!
interface Ethernet0/2
 description SIN-USO
 no ip address
 shutdown
 duplex auto
!
interface Ethernet0/3
 description SIN-USO
 no ip address
 shutdown
 duplex auto
!
ip forward-protocol nd
!
ip nat inside source list 10 interface Ethernet0/0 overload
ip route 0.0.0.0 0.0.0.0 200.7.42.1
ip ssh version 2
!
access-list 10 permit 10.7.42.0 0.0.0.127
!
banner motd ^CR-USUARIOS - Acceso restringido. Infra 3 - 2025-0742^C
!
line con 0
 password 7 <cifrada>
 logging synchronous
 login
line aux 0
line vty 0 4
 login local
 transport input ssh
!
end
