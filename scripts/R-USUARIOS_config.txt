! =====================================================================
! R-USUARIOS - Router Cisco IOL L3 - Infraestructura 3
! Jose Gabriel Feliz Maria - 2025-0742
! Sitio Usuarios: gateway y servidor DHCP de la VLAN 10 (10.7.42.0/25)
! y NAT (PAT) hacia el ISP. Claves de laboratorio.
! =====================================================================
enable
configure terminal
hostname R-USUARIOS
no ip domain-lookup
!
! --- Interfaces ---
interface Ethernet0/0
 description WAN-ISP (IP publica sitio Usuarios)
 ip address 200.7.42.2 255.255.255.252
 ip nat outside
 no shutdown
interface Ethernet0/1
 description LAN-USUARIOS (VLAN 10 via SW-A)
 ip address 10.7.42.1 255.255.255.128
 ip nat inside
 no shutdown
interface range Ethernet0/2 - 3
 description SIN-USO
 shutdown
exit
ip route 0.0.0.0 0.0.0.0 200.7.42.1
!
! --- DHCP para la VLAN 10 ---
ip dhcp excluded-address 10.7.42.1 10.7.42.9
ip dhcp pool USUARIOS-VLAN10
 network 10.7.42.0 255.255.255.128
 default-router 10.7.42.1
 dns-server 8.8.8.8 8.8.4.4
exit
!
! --- NAT hacia Internet ---
access-list 10 permit 10.7.42.0 0.0.0.127
ip nat inside source list 10 interface Ethernet0/0 overload
!
! --- Configuracion basica de seguridad ---
service password-encryption
enable secret cisco123
banner motd #R-USUARIOS - Acceso restringido. Infra 3 - 2025-0742#
ip domain-name itla.lab
username admin privilege 15 secret cisco123
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
line console 0
 password cisco
 login
 logging synchronous
line vty 0 4
 login local
 transport input ssh
end
write memory
